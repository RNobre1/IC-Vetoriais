% Contador de slides. A capa não recebe número, mas conta como slide 1 — é o
% que a apresentação anterior do grupo faz, em que o Roteiro aparece como 2.
\newcounter{slidenum}
\setcounter{slidenum}{1}

% Rodapé: logo do PET à esquerda, número do slide à direita.
\newcommand{\rodape}{%
  \vfill
  \hspace*{\margemesq}%
  \raisebox{-6pt}{\includegraphics[height=52pt]{logos/pet-rodape.png}}%
  \hfill
  {\color{cinza}\fontsize{22}{26}\selectfont \theslidenum}%
  \hspace*{56pt}\vspace*{24pt}%
}

% Um slide de conteúdo: título grande no alto à esquerda, corpo, rodapé.
% O corpo é alinhado à esquerda, não justificado: em fonte grande e linha
% larga, a justificação hifeniza palavra no meio e atravanca a leitura de
% quem está a metros da tela.
\newenvironment{slide}[1]{%
  \stepcounter{slidenum}%
  \vspace*{54pt}%
  \hspace*{\margemesq}%
  \begin{minipage}{\dimexpr1440pt-2\margemesq\relax}
  \raggedright
  {\fontsize{46}{54}\selectfont\bfseries #1}
  \end{minipage}\\[34pt]
  \hspace*{\margemesq}%
  \begin{minipage}{1296pt}
  \raggedright
  \fontsize{28}{42}\selectfont\color{black}
}{%
  \end{minipage}
  \rodape
  \newpage
}

% Cartão azul com rótulo branco, como no modelo. Altura mínima igual entre
% cartões vizinhos, mas livre para crescer se o texto pedir — a versão de
% altura fixa cortava a última linha sem avisar.
\newcommand{\cartao}[3]{% largura, título, linhas
  \begin{minipage}[t]{#1}
    \colorbox{azul}{\begin{minipage}[t][188pt][s]{\dimexpr#1-44pt\relax}
      \raggedright
      \vspace{16pt}
      {\color{white}\fontsize{26}{32}\selectfont\bfseries #2}\\[10pt]
      {\color{white}\fontsize{22}{29}\selectfont #3}
      \vfill
      \vspace{4pt}
    \end{minipage}}
  \end{minipage}%
}

% Número (ou palavra) em destaque, do tamanho de manchete, com legenda abaixo.
\newcommand{\numerao}[3]{% cor, destaque, legenda
  \begin{minipage}[t]{392pt}
    \raggedright
    {\color{#1}\fontsize{62}{70}\selectfont\bfseries #2}\\[14pt]
    {\color{cinza}\fontsize{25}{34}\selectfont #3}
  \end{minipage}%
}

% Barra horizontal proporcional, para o gráfico de tempo de indexação.
\newcommand{\barra}[4]{% rótulo, largura em pt, cor, valor escrito
  \makebox[190pt][l]{\fontsize{25}{30}\selectfont #1}%
  \textcolor{#3}{\rule[-4pt]{#2pt}{34pt}}%
  \hspace{14pt}{\fontsize{25}{30}\selectfont\bfseries #4}\\[20pt]
}

\newcommand{\dest}[1]{\textbf{\textcolor{azul}{#1}}}
